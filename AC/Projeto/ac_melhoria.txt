conf t
ip cef
mpls traffic-eng tunnels
class-map match-all MAP-L2
match ip dscp af31
class-map match-all MAP-L1
match ip dscp ef
class-map match-all L1
match access-group 101
class-map match-all L2
match access-group 102
policy-map POLICY
class MAP-L1
priority 10000
class MAP-L2
bandwidth 10000
policy-map MARK-PACKET
class L1
set ip dscp ef
class L2
set ip dscp af31

tunnel destination 10.0.0.11
tunnel mpls traffic-eng priority 7 7
tunnel mpls traffic-eng bandwidth 10000
tunnel mpls traffic-eng path-option 1 dynamic

int f0/1

no shut
ip ospf 1 area 0
mpls ip
mpls traffic-eng tunnels
service-policy output POLICY
ip rsvp bandwidth 20000 20000
int f1/0
ip address 10.0.2.5 255.255.255.252
no shut
ip ospf 1 area 0

end
